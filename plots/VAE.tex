\providecommand{\encoderFunc}{f_{\text{enc}}}
\providecommand{\decoderFunc}{f_{\text{dec}}}

\providecommand{\latentSpacePoint}{\mathbf{z}}
\providecommand{\dataPoint}{\mathbf{x}}
\providecommand{\dataPointEst}{\hat{\mathbf{x}}}

\begin{tikzpicture}[
    >=Stealth, 
    thick, 
    every node/.style={font=\Large}
]

    % --- COLORS ---
    \definecolor{myblue}{RGB}{200, 200, 245}
    \definecolor{mygrid}{RGB}{100, 100, 160}
    \definecolor{errorred}{RGB}{200, 50, 50}

    % --- LEFT SIDE: DATA SPACE (3D) ---
    \begin{axis}[
        at={(0,0)}, anchor=center,
        width=10cm, height=10cm,
        hide axis,
        view={40}{30}, 
        colormap={custom}{color(0cm)=(myblue); color(1cm)=(myblue)},
        z buffer=sort,
        name=plotarea
    ]
        
        % Surface
        \addplot3[
            surf,
            shader=faceted interp,
            fill opacity=0.7,
            faceted color=mygrid,
            domain=-1.6:1.6,
            y domain=-1.6:1.6,
            samples=19,
        ] {x^2 - y^2};

        % --- POINTS (Updated for closer proximity) ---
        
        % Point x: (-1.2, 0.3) -> z = 1.35 (High on the left wing)
        \coordinate (x_real) at (axis cs: -1.2, 0.3, 1.35); 
        
        % Point x_hat: Moved closer to x
        % x=-0.8 (closer to -1.2), y=0.7 (closer to 0.3)
        % z = (-0.8)^2 - (0.7)^2 = 0.64 - 0.49 = 0.15
        \coordinate (x_hat) at (axis cs: -0.8, 0.7, 0.15);

        % Draw the dots
        \addplot3[only marks, mark=*, mark size=3pt, color=black] coordinates {(-1.2, 0.3, 1.35)};
        \addplot3[only marks, mark=*, mark size=3pt, color=black] coordinates {(-0.8, 0.7, 0.15)};
        
        % Draw Error Line
        \draw[densely dotted, ultra thick, errorred] (x_real) -- (x_hat);
        
    \end{axis}

    % Labels
    \node[above] at (x_real) {$\dataPoint$};
    \node[below left] at (x_hat) {$\dataPointEst$};
    
    \node[anchor=south] at (0, 3.5) {\textsc{Data Space}};
    % \node[anchor=south] at (0, 3.0) {\Large $\mathbb{R}^n$};
    \node[anchor=south] at (0, 3.0) {\Large $\mathbb{R}^{d_x}$};
    % \node[mygrid] at (1, -2.5) {\Large $\mathcal{M}$};


    % --- RIGHT SIDE: LATENT SPACE (2D) ---
    \begin{scope}[shift={(6,-2)}]
        \draw[step=1, gray!20, thin] (0,0) grid (4,4);
        \draw[thick, ->] (0,0) -- (4.2,0);
        \draw[thick, ->] (0,0) -- (0,4.2);
        
        \coordinate (z_point) at (2.5, 2.5);
        \fill (z_point) circle (3pt) node[above] {$\latentSpacePoint$};
        
        \node at (2, 4.8) {\textsc{Latent Space}};
        \node at (2, 4.3) {\Large $\mathbb{R}^{d_z}$};
    \end{scope}


    % --- CONNECTIONS ---
    
    % Encoder: x -> z
    \draw[->, thick, shorten >= 5pt, shorten <= 5pt] (x_real) to[out=20, in=130] 
        node[midway, above, text=gray, name=encoder] {encoder} 
        node[midway, below] {\Large $\encoderFunc{}$} 
        (z_point);

    % Decoder: z -> x_hat
    \draw[->, thick, shorten >= 5pt, shorten <= 5pt] (z_point) to[out=230, in=-60] 
        node[midway, below=0.2cm, text=gray, name=decoder] {decoder} 
        node[midway, below=0.7cm] {\Large $\decoderFunc{} \approx \encoderFunc{}^{-1}$} 
        (x_hat);

    % % Approximation Text (Centered)
    % \node[blue!50!gray] at ($(encoder)!.5!(decoder)$) {\Large $\decoderFunc{} \approx \encoderFunc{}^{-1}$};

\end{tikzpicture}
