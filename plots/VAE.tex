\providecommand{\encoderFunc}{f_{\text{enc}}}
\providecommand{\decoderFunc}{f_{\text{dec}}}

\providecommand{\latentSpacePoint}{\mathbf{z}}

\providecommand{\dataPoint}{\mathbf{x}}
\providecommand{\dataPointEst}{\hat{\mathbf{x}}}

\providecommand{\zMean}{\boldsymbol{\mu}_\phi}
\providecommand{\zVar}{\boldsymbol{\sigma}^2_\phi}
\providecommand{\xMean}{\boldsymbol{g}_\theta}

\begin{tikzpicture}[
    >=Stealth, 
    thick, 
    every node/.style={font=\Large}
]

    % --- COLORS ---
    % \definecolor{myblue}{RGB}{200, 200, 245}
    % \definecolor{mygrid}{RGB}{100, 100, 160}
    \definecolor{myblue}{RGB}{235, 235, 255}
    \definecolor{mygrid}{RGB}{130, 130, 190}

    \definecolor{errorred}{RGB}{200, 50, 50}
    \definecolor{pdfcolor}{RGB}{100, 100, 180}

    % --- LEFT SIDE: DATA SPACE (3D) ---
    \begin{axis}[
        at={(0,0)}, anchor=center,
        width=10cm, height=10cm,
        hide axis,
        view={40}{30}, 
        colormap={custom}{color(0cm)=(myblue); color(1cm)=(myblue)},
        z buffer=sort,
        name=plotarea
    ]
        
        % Surface
        \addplot3[
            surf,
            shader=faceted interp,
            fill opacity=0.7,
            faceted color=mygrid,
            domain=-1.6:1.6,
            y domain=-1.6:1.6,
            samples=19,
        ] {x^2 - y^2};

        % --- POINTS ---
        % Point x (Original)
        \coordinate (x_real) at (axis cs: -1.2, 0.3, 1.35); 
        
        % Point x_hat (Reconstruction) - MOVED CLOSER
        % x = -1.0 (closer to -1.2)
        % y = 0.5  (closer to 0.3)
        % z = (-1.0)^2 - (0.5)^2 = 0.75
        \coordinate (x_hat) at (axis cs: -1.0, 0.5, 0.75);

        % Draw the dots
        \addplot3[only marks, mark=*, mark size=3pt, color=black] coordinates {(-1.2, 0.3, 1.35)};
        \addplot3[only marks, mark=*, mark size=3pt, color=black] coordinates {(-1.0, 0.5, 0.75)};
        
        % Draw Error Line
        \draw[densely dotted, ultra thick, errorred] (x_real) -- (x_hat);
        
    \end{axis}

    % Labels
    % \node[above] at (x_real) {$\dataPoint$};
    % \node[below right] at (x_hat) {$\dataPointEst$};

    \node[above left] at (x_real) {$\dataPoint$};
    \node[below left=0.1em and 0.3em,outer sep=0, inner sep=0] at (x_hat) {$\dataPointEst$};
    
    \node[anchor=south] at (0, 3.5) {\textsc{Data Space}};
    \node[anchor=south] at (0, 3.0) {\Large $\mathbb{R}^{d_x}$};


    % --- RIGHT SIDE: LATENT SPACE (2D) ---
    \begin{scope}[shift={(8,0)}] 
        
        % 1. Draw Grid and Axes (Centered at 0,0)
        \draw[step=1, gray!20, thin] (-2.2,-2.2) grid (2.2,2.2);
        \draw[thick, ->] (-2.5,0) -- (2.5,0);
        \draw[thick, ->] (0,-2.5) -- (0,2.5);
        
        % 2. Gaussian Contour Plot (Standard Normal centered at 0,0)
        \begin{axis}[
            at={(0,0)}, anchor=center,
            width=5.5cm, height=5.5cm, scale only axis, 
            hide axis,
            xmin=-2.5, xmax=2.5, ymin=-2.5, ymax=2.5,
            view={0}{90},
            colormap={gaussian_map}{color(0cm)=(white); color(1cm)=(pdfcolor)},
            fill opacity=0.6,
        ]
            \addplot3[
                contour filled={number=9, labels=false}, 
                domain=-2.5:2.5,
                y domain=-2.5:2.5,
                samples=40, 
            ] 
            {exp(-(x^2 + y^2) / (2 * 0.6^2))};
        \end{axis}

        % 3. Point z (Close to mean 0,0)
        \coordinate (z_point) at (0.3, 0.3);
        \fill (z_point) circle (3pt) node[above right] {$\latentSpacePoint$};
        
        % Labels
        \node at (0, 3.2) {\textsc{Latent Space}};
        \node at (0, 2.7) {\Large $\mathbb{R}^{d_z}$};
    \end{scope}


    % --- CONNECTIONS ---
    
    % Encoder: x -> z
    \draw[->, thick, shorten >= 5pt, shorten <= 5pt] (x_real) to[out=20, in=150] 
        node[midway, above=1ex, text=gray, name=encoder] {encoder} 
        % node[midway, below] {\Large $\encoderFunc{}$} 
        node[midway, below] {\Large $\zMean,\zVar$} 
        (z_point);

    % Decoder: z -> x_hat
    \draw[->, thick, shorten >= 5pt, shorten <= 5pt] (z_point) to[out=220, in=-20] 
        node[midway, below=0.2cm, text=gray, name=decoder] {decoder} 
        % node[midway, below=0.7cm] {\Large $\decoderFunc{} \approx \encoderFunc{}^{-1}$} 
        node[midway, below=0.7cm] {\Large $\xMean$} 
        (x_hat);

\end{tikzpicture}
