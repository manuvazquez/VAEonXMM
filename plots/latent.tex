\documentclass{standalone}
\usepackage{pgfplots}

\usetikzlibrary{fit,positioning}

\pgfplotsset{compat=1.18}
\pgfplotsset{plot coordinates/math parser=false,unbounded coords=discard}

\providecommand{\dataDir}{paper_main}
\providecommand{\zoomSubdirs}{east,north}

\begin{document}
\begin{tikzpicture}
  \begin{axis}[
    xlabel={t-SNE${}_1$},
    ylabel={t-SNE${}_2$},
    title={Latent space projection (t-SNE)},
    grid=major,
    clip=false,
    scatter/classes={
      Unlabeled={
        mark=o,
        draw=gray,
        draw opacity=0.05,
        fill=none,
      },
      {NO Event}={
        mark=diamond*,
        fill=red,
      },
      Event={
        mark=square*,
        fill=green
      }
    },
  legend style={
    at={(0.5,-0.2)}, % adjust position
    anchor=north,     % attach legend’s top to that point
    legend columns=-1 % put all entries in one horizontal row
  },
  ]

    % one pass: use the "Color" column to pick class styles
    \addplot[
      scatter,
      forget plot,
      only marks,
      scatter src=explicit symbolic,
    ]
    table[
      x=tsne_1, y=tsne_2,
      meta=Color,
      col sep=comma,
      trim cells=true,
      ignore chars={ },
    ]{\dataDir/tSNE.csv};

    % ---------------- Legend entries ----------------


  \pgfplotsset{legend image post style={xshift=2pt}} % keeps hollow circle off the frame

  \addlegendimage{
    only marks,
    mark=o, mark size=3pt,
    mark options={draw=gray, line width=0.8pt, fill=none}
  }
  \addlegendimage{
    only marks,
    mark=diamond*, mark size=3pt,
    mark options={draw=red, fill=red}
  }
  \addlegendimage{
    only marks,
    mark=square*, mark size=3pt,
    mark options={draw=green, fill=green}
  }

  % \legend{Unlabeled, No Event, Event}

  \addlegendimage{
    only marks,
    mark=star, 
    mark size=2.2pt,
    draw=blue
  }

  % Update the list of labels to include the 4th item
  \legend{Unlabeled, No Event, Event, Plotted}

  % if zoom subdirs (comma-separated list like "-8-9_-6-6,-95--84_58-66") have been passed...  
  \ifdefined\zoomSubdirs

    % \def\zoomNodeAnchors{north east, north, north west, south}
    \def\zoomNodeAnchors{north east, north, east, south west, south west}

    % for each subdir...
    \foreach \subdir [
      count=\iSubdir from 1,
    ] in \zoomSubdirs {

      % in order to "locate" the anchor for this index (Gemini)
      \foreach \anc [count=\j from 1] in \zoomNodeAnchors {
          \ifnum\j=\iSubdir
              \xdef\currentAnchor{\anc} % \xdef makes it global to escape the inner loop
          \fi
      }
      
      % CSV with the coordinates of the zoom rectangle are read
      \pgfplotstableread[col sep=comma]{\dataDir/\subdir/coordinates.csv}\zoombox

      % corners are parsed from there
      \pgfplotstablegetelem{0}{x}\of\zoombox \edef\xa{\pgfplotsretval}
      \pgfplotstablegetelem{0}{y}\of\zoombox \edef\ya{\pgfplotsretval}
      \pgfplotstablegetelem{1}{x}\of\zoombox \edef\xb{\pgfplotsretval}
      \pgfplotstablegetelem{1}{y}\of\zoombox \edef\yb{\pgfplotsretval}

      % \typeout{[zoom \iSubdir] \subdir: xa=\xa, ya=\ya, xb=\xb, yb=\yb}

      % rectangle is plotted...
      \expanded{\noexpand\draw[black, very thick] (axis cs:\xa,\ya) rectangle (axis cs:\xb,\yb);}

      \expanded{%
        \noexpand\node[
          fit={(axis cs:\xa,\ya) (axis cs:\xb,\yb)},
          % inner sep=0pt,
          % outer sep=0pt,
        ] (zoombox\iSubdir) {};%
      }

      % ...along with a "label"
      % \expanded{\noexpand\node[right] at (axis cs:\xb,\yb) {\#\iSubdir};}
      \expanded{%
        \noexpand
        \node[
        ] at (zoombox\iSubdir.\currentAnchor) {\#\iSubdir};
      }

      % highlighted points for this region are plotted
      \addplot+[
        only marks,
        mark=star,
        mark size=2.2pt,
        draw=blue,
      ] table[
        x=tsne_1, y=tsne_2,
        col sep=comma,
      ]{\dataDir/\subdir/selected_windows.csv};
    }

  \fi

  \end{axis}
\end{tikzpicture}

\end{document}
